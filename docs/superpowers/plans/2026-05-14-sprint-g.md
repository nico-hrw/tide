Update `handleSwipeMove` to check vertical pull at top:
```tsx
const handleSwipeMove = (e: React.TouchEvent) => {
  const deltaX = e.touches[0].clientX - swipeTouchStartX.current;
  setSwipeDelta(deltaX);
  const idx = tabOrder.indexOf(activeTab);
  if (idx === 2 && deltaX < 0) {
    const progress = Math.min(1, Math.abs(deltaX) / 150);
    setTrackerProgress(progress);
    trackerProgressMV.set(progress);
  } else {
    setTrackerProgress(0);
    trackerProgressMV.set(0);
  }

  // Vertical pull: expand month calendar only when pulling down hard from top
  if (activeTab === 'calendar' && !isMonthExpanded) {
    const el = e.currentTarget as HTMLElement;
    if (el.scrollTop <= 0) {
      const pullDown = e.touches[0].clientY - verticalPullStartY.current;
      if (pullDown > 60) {
        setIsMonthExpanded(true);
      }
    }
  }
};
```
