Read the full file first.

- [ ] **Step 1: Add swipe tracking state and refs**

Inside `MobileLayout`, after existing state declarations, add:

```tsx
// Swipe navigation
const tabOrder: Tab[] = ['notes', 'calendar', 'profile'];
const activeTabIndex = tabOrder.indexOf(activeTab);
const swipeTouchStartX = useRef(0);
const [swipeDelta, setSwipeDelta] = useState(0);        // px offset during drag
const [trackerProgress, setTrackerProgress] = useState(0); // 0-1 for circular indicator
const [swipeDir, setSwipeDir] = useState<1 | -1>(1);    // 1=left→right, -1=right→left
```

- [ ] **Step 2: Add swipe handlers**

After `handleScroll`, add:

```tsx
const handleSwipeStart = (e: React.TouchEvent) => {
  swipeTouchStartX.current = e.touches[0].clientX;
  setSwipeDelta(0);
  setTrackerProgress(0);
};

const handleSwipeMove = (e: React.TouchEvent) => {
  const delta = e.touches[0].clientX - swipeTouchStartX.current;
  setSwipeDelta(delta);

  // Tracker resistance: when on Profile (index 2) dragging left
  if (activeTabIndex === 2 && delta < 0) {
    const progress = Math.min(1, Math.abs(delta) / 150);
    setTrackerProgress(progress);
  } else {
    setTrackerProgress(0);
  }
};

const handleSwipeEnd = () => {
  const delta = swipeDelta;
  const threshold = 60;

  if (delta < -threshold) {
    // Swipe left → go to next tab (or tracker)
    if (activeTabIndex === 2) {
      // Tracker: only navigate if progress reached 100%
      if (trackerProgress >= 1) {
        const url = process.env.NEXT_PUBLIC_TRACKER_URL;
        if (url) window.location.href = url;
      }
    } else {
      setSwipeDir(1);
      setActiveTab(tabOrder[Math.min(activeTabIndex + 1, 2)]);
    }
  } else if (delta > threshold) {
    // Swipe right → go to prev tab
    if (activeTabIndex > 0) {
      setSwipeDir(-1);
      setActiveTab(tabOrder[Math.max(activeTabIndex - 1, 0)]);
    }
  }

  setSwipeDelta(0);
  setTrackerProgress(0);
};
```
